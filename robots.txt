# =====================================================================
# SARMAD Attendance - crawler policy (STAGE 22)
# =====================================================================
# The public landing page (the site root) is the ONLY page that must be
# indexed. Everything else is the application surface: the login screen,
# the manager panel, the owner panel, the device kiosk (a PWA that lives
# on a tablet), the JSON API, the installer and the internal folders of
# the package.
#
# How this is resolved: a crawler uses the LONGEST matching pattern, so
# "Allow: /$" (2 characters) beats the catch-all "Disallow: /" (1
# character) for the root URL, while every other URL stays blocked.
# The assets folder is allowed on purpose - Googlebot must be able to
# fetch landing.css and landing.js to render and judge the page.
#
# Sub-directory installation: this file is only ever served at the domain
# root (public_html/robots.txt). If the package is extracted into a folder
# such as public_html/attendance/, point the Allow lines at that folder:
#     Allow: /attendance/$
#     Allow: /attendance/assets/
# =====================================================================

User-agent: *

# --- Public marketing page: open to every crawler ---------------------
Allow: /$
Allow: /assets/
Allow: /manifest.webmanifest
Allow: /robots.txt

# --- Device kiosk (PWA on a tablet, holds a device token) -------------
Disallow: /device
Disallow: /device/

Disallow: /logout

Disallow: /install/
